% Preamble: packages, page layout and macros of the thesis.
% No official university template exists (author, 2026-10-16), so the layout follows common thesis practice:
% A4, 12 pt, 1.5 line spacing, a wider binding margin.

\usepackage[a4paper,top=2.5cm,bottom=2.5cm,inner=3cm,outer=2.5cm]{geometry}
% Calibri (author's choice, 2026-10-16) needs a Unicode engine: compile with XeLaTeX or LuaLaTeX.
% Calibri ships with Windows and Office; where it is missing (Overleaf, Linux) Carlito is used, a free font
% with the same metrics, so line and page breaks stay the same.
\usepackage{fontspec}
% English hyphenation patterns; without them XeLaTeX cannot break long words and lines overflow the margin.
\usepackage[english]{babel}
\IfFontExistsTF{Calibri}{\setmainfont{Calibri}}{\setmainfont{Carlito}}
\usepackage{microtype}
% Allow a little extra inter-word space before a line overflows, for paragraphs with long names and citations.
\setlength{\emergencystretch}{1.5em}
% Long DOIs and URLs in the bibliography may break at any character.
\usepackage{xurl}
\usepackage{setspace}
\onehalfspacing

\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{multirow}
\usepackage{siunitx}
\usepackage{xcolor}
\usepackage{caption}
\usepackage{subcaption}
\usepackage{enumitem}
\usepackage{listings}

\usepackage[style=ieee,backend=biber]{biblatex}
\addbibresource{refs.bib}

% noidx mode: the list is sorted by LaTeX itself, so no external indexing tool is needed.
\usepackage[acronym,toc]{glossaries}
\makenoidxglossaries

\usepackage[hidelinks]{hyperref}
\usepackage[nameinlink,capitalise]{cleveref}

% Figures are drawn by figures/scripts/ (repository root); the thesis only includes them.
\graphicspath{{../figures/results/}{../figures/diagrams/}}

% A framed box standing in for a figure that is drawn in a later stage, so the text compiles before it exists;
% \figplaceholder{plan id}{what the figure will show}. None may remain in the submitted thesis (Stage 8 check).
\newcommand{\figplaceholder}[2]{\fbox{\parbox{0.9\linewidth}{\centering\vspace{2cm}\textbf{[#1]} #2\vspace{2cm}}}}

% A visible red note for a statement that is still provisional, e.g. an attribution not yet confirmed.
% None may remain in the submitted thesis (Stage 8 check).
\newcommand{\checknote}[1]{\textcolor{red}{\,[\textbf{to check:} #1]}}

% A proportion with its denominator, e.g. \kn{195}{200}; every rate in the text is written this way.
\newcommand{\kn}[2]{#1/#2}
